### 星星还是树

题目：https://www.acwing.com/problem/content/3170/

```c++
#include <algorithm>
#include <cmath>
#include <cstdio>
#include <cstdlib>
#include <ctime>
#include <iostream>

const int N = 110;

int n;

struct Point {
    double x, y;
} P[N];

double res;

double rand(double l, double r) {
    return double(rand()) / RAND_MAX * (r - l) + l;
}

double getDist(const Point& a, const Point& b) {
    double dx = a.x - b.x, dy = a.y - b.y;
    return sqrt(dx * dx + dy * dy);
}

double getDist(const Point& p) {
    double dist = 0;
    for (int i = 0; i < n; i++) {
        dist += getDist(p, P[i]);
    }
    res = std::min(res, dist);
    return dist;
}

void sa() {
    Point cur{rand(0, 1e4), rand(0, 1e4)};
    for (double t = 1e4; t > 1e-4; t *= 0.9) {
        Point nxt{rand(cur.x - t, cur.x + t), rand(cur.y - t, cur.y + t)};
        double d = getDist(nxt) - getDist(cur);
        if (exp(-d / t) > rand(0, 1)) {
            cur = nxt;
        }
    }
}

int main() {
    std::cin >> n;
    for (int i = 0; i < n; i++) {
        int x, y;
        std::cin >> x >> y;
        P[i] = {double(x), double(y)};
    }
    res = 1e8;
    for (int i = 0; i < 100; i++) {
        sa();
    }
    printf("%.0lf\n", res);
    return 0;
}
```

### 保龄球

题目：https://www.acwing.com/problem/content/description/2426/

```c++
#include <algorithm>
#include <cmath>
#include <cstdlib>
#include <iostream>

const int N = 55;

int n, m;
int X[N], Y[N];
int res;

void swap(int a, int b) {
    std::swap(X[a], X[b]);
    std::swap(Y[a], Y[b]);
}

int calc() { // 计算能量值，注意能量值是越小越好
    int score = 0;
    for (int i = 0; i < m; i++) {
        score += X[i] + Y[i];
        if (i < m) {
            if (X[i] == 10) {
                score += X[i + 1] + Y[i + 1];
            } else if (X[i] + Y[i] == 10) {
                score += X[i + 1];
            }
        }
    }
    res = std::max(res, score);
    return -score;
}

void simulatedAnneal() {
    for (double t = 1e4; t > 1e-4; t *= 0.99) {
        int a = rand() % m, b = rand() % m;
        int cur = calc();
        swap(a, b);
        if (n + (X[n - 1] == 10) == m) {
            int nxt = calc();
            double d = nxt - cur;
            if (exp(-d / t) < double(rand()) / RAND_MAX) {
                swap(a, b);
            }
        } else {
            swap(a, b);
        }
    }
}

int main() {
    std::cin >> n;
    m = n;
    for (int i = 0; i < n; i++) {
        std::cin >> X[i] >> Y[i];
    }
    if (X[n - 1] == 10) {
        std::cin >> X[n] >> Y[n];
        m++;
    }
    for (int i = 0; i < 100; i++) {
        simulatedAnneal();
    }
    std::cout << res << '\n';
    return 0;
}
```

### 均分数据

题目：https://www.acwing.com/problem/content/2682/

```c++
#include <algorithm>
#include <cmath>
#include <cstdlib>
#include <iomanip>
#include <iostream>
#include <numeric>

const int N = 25, M = 10;

int n, m;
int A[N], S[M];
double res{1e8};

double calc() {
    std::fill(S, S + M, 0);
    for (int i = 0; i < n; i++) {
        auto it = std::min_element(S, S + m);
        *it += A[i];
    }
    double avg = (double) std::reduce(S, S + M, 0) / m;
    double delta = 0;
    for (int i = 0; i < m; i++) {
        delta += (S[i] - avg) * (S[i] - avg);
    }
    delta = sqrt(delta / m);
    res = std::min(res, delta);
    return delta;
}

void simulatedAnneal() {
    for (double t = 1e6; t > 1e-6; t *= 0.95) {
        double cur = calc();
        int a = rand() % n, b = rand() % n;
        std::swap(A[a], A[b]);
        double nxt = calc();
        double d = nxt - cur;
        if (exp(-d / t) < double(rand()) / RAND_MAX) {
            std::swap(A[a], A[b]);
        }
    }
}

int main() {
    std::cin >> n >> m;
    for (int i = 0; i < n; i++) {
        std::cin >> A[i];
    }
    for (int i = 0; i < 100; i++) {
        simulatedAnneal();
    }
    std::cout << std::fixed << std::setprecision(2) << res << '\n';
    return 0;
}
```
