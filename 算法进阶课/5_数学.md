## 莫比乌斯反演

### promblem b

题目：https://www.acwing.com/problem/content/2704/

题解：https://www.acwing.com/solution/content/64817/

```c++
#include <cstdio>
#include <iostream>
#include <algorithm>
#include <numeric>

using LL = long long;
using std::cin, std::cout;

const int N = int(5e4) + 5;

int a, b, c, d, k;
bool st[N];
int prime[N], cnt;
int mob[N], ps[N];

void mobiusSieve()
{
    mob[1] = 1;
    for (int i = 2; i < N; i++)
    {
        if (!st[i])
        {
            prime[cnt++] = i;
            mob[i] = -1;
        }
        for (int j = 0; LL(prime[j]) * i < N; j++)
        {
            st[prime[j] * i] = true;
            if (i % prime[j] == 0) break;
            mob[prime[j] * i] = -mob[i];
        }
    }
}

int endPoint(int i, int n)
{
    return n / (n / i);
}

LL f(int a, int b)  // (0,0)->(a,b)矩形内的所有满足要求的点
{
    a /= k, b /= k;
    int n = std::min(a, b);
    LL res = 0;
    for (int l = 1, r = 1; l <= n; l = r + 1)
    {
        r = std::min({ n, endPoint(l, a), endPoint(l, b) });
        res += LL(ps[r] - ps[l - 1]) * (a / l) * (b / l);
    }
    return res;
}

void solve()
{
    cin >> a >> b >> c >> d >> k;
    cout << f(b, d) - f(b, c - 1) - f(a - 1, d) + f(a - 1, c - 1) << '\n';
}

int main()
{
    std::ios::sync_with_stdio(false);
    cin.tie(nullptr), cout.tie(nullptr);

    mobiusSieve();
    for (int i = 1; i < N; i++) ps[i] = ps[i - 1] + mob[i];

    int t; cin >> t;
    while (t--)
        solve();

    return 0;
}
```

### 约数个数和

题目：https://www.acwing.com/problem/content/description/1360/

题解：https://www.acwing.com/solution/content/172178/

```c++
#include <cstdio>
#include <iostream>
#include <algorithm>
#include <numeric>

using LL = long long;
using std::cin, std::cout;

const int N = int(5e4) + 5;

bool st[N];
int prime[N], cnt;
int mob[N], ps[N];
int h[N];

void init()
{
    mob[1] = 1;
    for (int i = 2; i < N; i++)
    {
        if (!st[i])
        {
            prime[cnt++] = i;
            mob[i] = -1;
        }
        for (int j = 0; LL(prime[j]) * i < N; j++)
        {
            st[prime[j] * i] = true;
            if (i % prime[j] == 0) break;
            mob[prime[j] * i] = -mob[i];
        }
    }
    std::partial_sum(mob + 1, mob + 1 + N, ps + 1);

    for (int i = 1; i < N; i++)
    {
        for (int l = 1, r; l <= i; l = r + 1)
        {
            r = std::min(i, i / (i / l));  // 分块
            h[i] += (r - l + 1) * (i / l);
        }
    }
}

int n, m, mn;

void solve()
{
    cin >> n >> m;
    mn = std::min(n, m);

    LL res = 0;
    for (int l = 1, r; l <= mn; l = r + 1)
    {
        r = std::min({ mn, n / (n / l) , m / (m / l) });
        res += LL(ps[r] - ps[l - 1]) * h[n / l] * h[m / l];
    }
    cout << res << '\n';
}

int main()
{
    std::ios::sync_with_stdio(false);
    cin.tie(nullptr), cout.tie(nullptr);

    init();

    int t; cin >> t;
    while (t--)
        solve();

    return 0;
}
```

## 积性函数

### 龙哥的问题

题目：https://www.acwing.com/problem/content/223/

题解：https://www.acwing.com/solution/content/159967/

```c++
#include <cstdio>
#include <iostream>
#include <algorithm>
#include <numeric>

using LL = long long;
using std::cin, std::cout;

int phi(int x)  // 求欧拉函数的值
{
    int res = x;
    for (int i = 2; i <= x / i; i++)
        if (x % i == 0)
        {
            res = res * LL(i - 1) / i;
            while (x % i == 0) x /= i;
        }
    if (x > 1) res = res * LL(x - 1) / x;
    return res;
}

int main()
{
    std::ios::sync_with_stdio(false);
    cin.tie(nullptr), cout.tie(nullptr);

    int n; cin >> n;
    LL res = 0;
    for (int i = 1; i <= n / i; i++)
        if (n % i == 0)
        {
            res += i * phi(n / i);
            if (i != n / i) res += (n / i) * phi(i);
        }
    cout << res << '\n';

    return 0;
}
```

## BSGS

### BSGS

题目：https://www.acwing.com/problem/content/3127/

题解：https://www.acwing.com/solution/content/79890/

```c++
#include <cmath>
#include <iostream>
#include <unordered_map>

using LL = long long;

int bsgs(int a, int p, int b) {
	if (1 % p == b % p) {
		return 0;
	}
	int k = sqrt(p) + 1;
	std::unordered_map<int, int> ht;
	for (int i = 0, j = b % p; i < k; i++) {
		ht[j] = i;
		j = LL(j) * a % p;
	}
	int ak = 1;
	for (int i = 1; i <= k; i++) {
		ak = LL(ak) * a % p;
	}
	for (int i = 1, j = ak; i <= k; i++) {
		auto it = ht.find(j);
		if (it != ht.end()) {
			return LL(i) * k - it->second;
		}
		j = LL(j) * ak % p;
	}
	return -1;
}

int main() {

	int a, b, p;
	while (std::cin >> a >> p >> b, a || p || b) {
		int ret = bsgs(a, p, b);
		if (ret < 0) {
			std::cout << "No Solution" << '\n';
		}
		else {
			std::cout << ret << '\n';
		}
	}
	return 0;
}
```

### 扩展BSGS

题目：https://www.acwing.com/problem/content/3128/

```c++
#include <cmath>
#include <iostream>
#include <unordered_map>

using LL = long long;

int exgcd(int a, int b, int& x, int& y) {
	if (!b) {
		x = 1, y = 0;
		return a;
	}
	int d = exgcd(b, a % b, y, x);
	y -= a / b * x;
	return d;
}

int bsgs(int a, int p, int b) {
	if (1 % p == b % p) {
		return 0;
	}
	int k = sqrt(p) + 1;
	std::unordered_map<int, int> ht;
	for (int i = 0, j = b % p; i < k; i++) {
		ht[j] = i;
		j = LL(j) * a % p;
	}
	int ak = 1;
	for (int i = 1; i <= k; i++) {
		ak = LL(ak) * a % p;
	}
	for (int i = 1, j = ak; i <= k; i++) {
		auto it = ht.find(j);
		if (it != ht.end()) {
			return LL(i) * k - it->second;
		}
		j = LL(j) * ak % p;
	}
	return -1;
}

int exbsgs(int a, int p, int b) {
	b = (LL(b) % p + p) % p;
	if (1 % p == b % p) {
		return 0;
	}
	int x = 0, y = 0;
	int d = exgcd(a, p, x, y);
	if (d > 1) {
		if (b % d) {
			return -1;
		}
		exgcd(a / d, p / d, x, y);
		int ret = exbsgs(a, p / d, LL(x) * (b / d) % (p / d));
		if (ret == -1) {
			return -1;
		}
		return ret + 1;
	}
	return bsgs(a, p, b);
}

int main() {
	int a, p, b;
	while (std::cin >> a >> p >> b, a || p || b) {
		int ret = exbsgs(a, p, b);
		if (ret == -1) {
			std::cout << "No Solution" << '\n';
		}
		else {
			std::cout << ret << '\n';
		}
	}
	return 0;
}
```

### 随机数生成器

题目：https://www.acwing.com/problem/content/description/2528/

题解：https://www.acwing.com/solution/content/172906/

```c++
#include <cmath>
#include <iostream>
#include <unordered_map>

using LL = long long;

int exgcd(int a, int b, int& x, int& y) {
	if (!b) {
		x = 1, y = 0;
		return a;
	}
	int d = exgcd(b, a % b, y, x);
	y -= a / b * x;
	return d;
}

int mod(int a, int p) {
	return (a % p + p) % p;
}

int inv(int a, int p) {
	int x = 0, y = 0;
	exgcd(a, p, x, y);
	return mod(x, p);
}

int bsgs(int a, int p, int b) {
	if (1 % p == b % p) {
		return 0;
	}
	int k = sqrt(p) + 1;
	std::unordered_map<int, int> ht;
	for (int i = 0, j = b % p; i < k; i++) {
		ht[j] = i;
		j = LL(j) * a % p;
	}
	int ak = 1;
	for (int i = 1; i <= k; i++) {
		ak = LL(ak) * a % p;
	}
	for (int i = 1, j = ak; i <= k; i++) {
		auto it = ht.find(j);
		if (it != ht.end()) {
			return i * k - it->second;
		}
		j = LL(j) * ak % p;
	}
	return -1;
}

void solve() {
	int p, a, b, x1, t; std::cin >> p >> a >> b >> x1 >> t;
	if (a == 0) {
		if (t == x1) {
			std::cout << 1 << '\n';
		}
		else if (t == b) {
			std::cout << 2 << '\n';
		}
		else {
			std::cout << -1 << '\n';
		}
	}
	else if (a == 1) {
		if (b == 0) {
			if (t == x1) {
				std::cout << 1 << '\n';
			}
			else {
				std::cout << -1 << '\n';
			}
		}
		else {
			int n = LL(mod(t - x1, p)) * inv(b, p) % p;
			std::cout << ++n << '\n';
		}
	}
	else {
		int c = LL(b) * inv(a - 1, p) % p;
		if ((x1 + c) % p == 0) {
			if (t == mod(-c, p)) {
				std::cout << 1 << '\n';
			}
			else {
				std::cout << -1 << '\n';
			}
		}
		else {
			int r = LL(t + c) * inv(x1 + c, p) % p;
			int ret = bsgs(a, p, r);
			if (ret == -1) {
				std::cout << -1 << '\n';
			}
			else {
				std::cout << ret + 1 << '\n';
			}
		}
	}
}

int main() {
	int t; std::cin >> t;
	while (t--) {
		solve();
	}
	return 0;
}
```

## FFT

### 多项式乘法

题目：https://www.acwing.com/problem/content/3125/

题解：https://www.acwing.com/solution/content/125076/

```c++
#include <cmath>
#include <iostream>
#include <algorithm>

const int N = (1 << 18) + 10;
const double PI = acos(-1);

struct Complex {
	double x, y;
	Complex operator+(const Complex& o) const {
		return { x + o.x, y + o.y };
	}
	Complex operator-(const Complex& o) const {
		return { x - o.x, y - o.y };
	}
	Complex operator*(const Complex& o) const {
		return { x * o.x - y * o.y, x * o.y + y * o.x };
	}
};

Complex A[N], B[N];
int bit, tot;
int rev[N];

void FFT(Complex A[], int type) {
	for (int i = 0; i < tot; i++) {
		if (i < rev[i]) {
			std::swap(A[i], A[rev[i]]);
		}
	}
	for (int mid = 1; mid < tot; mid *= 2) {
		Complex w1 = { cos(PI / mid), type * sin(PI / mid) };
		for (int i = 0; i < tot; i += mid * 2) {
			Complex wk{ 1, 0 };
			for (int j = 0; j < mid; j++, wk = wk * w1) {
				auto x = A[i + j], y = wk * A[i + mid + j];
				A[i + j] = x + y;
				A[i + mid + j] = x - y;
			}
		}
	}
}

int main() {
	int n, m; std::cin >> n >> m;
	for (int i = 0; i <= n; i++) {
		std::cin >> A[i].x;
	}
	for (int i = 0; i <= m; i++) {
		std::cin >> B[i].x;
	}
	while ((n + m + 1) >> bit) {
		bit++;
	}
	tot = 1 << bit;
	for (int i = 1; i < tot; i++) {
		rev[i] = (rev[i >> 1] >> 1) | ((i & 1) << (bit - 1));
	}
	FFT(A, 1), FFT(B, 1);
	for (int i = 0; i < tot; i++) {
		A[i] = A[i] * B[i];
	}
	FFT(A, -1);
	for (int i = 0; i <= n + m; i++) {
		std::cout << int(A[i].x / tot + 0.5) << ' ';
	}
	std::cout << '\n';

	return 0;
}
```

```c++
		A[i].x = s1[n - i] - '0';

	}
```

```c++
const int P = 10007;

int FP(int a, int b) {
	int res = 1;

		if (b & 1) {
			res = res * a % P;
		}
		a = a * a % P;
		b >>= 1;
	}
	return res;
```

```c++
#include <iostream>

int fp(int a, int b) {
	int res = 1;
	while (b) {
		if (b & 1) {
			res = res * a;
		}
		a = a * a;
		b >>= 1;
	}
	return res;

	if (!b) {
		return a;
	}
	return gcd(b, a % b);

		int res = 0;

			res += n * fp(m, (n + 1) / 2);
		}
		else {
			int h = n / 2;

		}
		res /= n * 2;

	}
```

### 魔法手链

题目：https://www.acwing.com/problem/content/3137/

题解：https://www.acwing.com/solution/content/244152/

```c++
#include <iostream>
#include <algorithm>

const int N = 11, P = 9973;

struct Matrix {
	int n;
	int val[N][N];
	Matrix(int v, int n) : n{ n } {
		std::fill_n((int*)val, N * N, v);
	}
	Matrix operator* (const Matrix& o) const {
		Matrix res{ 0, n };
		for (int i = 1; i <= n; i++) {
			for (int j = 1; j <= n; j++) {
				for (int k = 1; k <= n; k++) {
					res.val[i][j] = (res.val[i][j] + val[i][k] * o.val[k][j]) % P;
				}
			}
		}
		return res;
	}
};

Matrix fp(Matrix a, int b) {
	Matrix res{ 0, a.n };
	for (int i = 1; i <= a.n; i++) {
		res.val[i][i] = 1;
	}
	while (b) {
		if (b & 1) {
			res = res * a;
		}
		a = a * a;
		b >>= 1;
	}
	return res;
}

int dp(const Matrix& a, int d) {
	Matrix M = fp(a, d);
	int res = 0;
	for (int i = 1; i <= a.n; i++) {
		res = (res + M.val[i][i]) % P;
	}
	return res;
}

int phi(int n) {
	int res = n;
	for (int i = 2; i <= n / i; i++) {
		if (n % i == 0) {
			res = res / i * (i - 1);
			while (n % i == 0) {
				n /= i;
			}
		}
	}
	if (n > 1) {
		res = res / n * (n - 1);
	}
	return res % P;
}

int fp(int a, int b) {
	int res = 1;
	a %= P;
	while (b) {
		if (b & 1) {
			res = res * a % P;
		}
		a = a * a % P;
		b >>= 1;
	}
	return res;
}

int inv(int a) {
	return fp(a, P - 2);
}

void solve() {
	int n, m, k; std::cin >> n >> m >> k;
	Matrix tr{ 1, m };
	for (int i = 0; i < k; i++) {
		int a, b; std::cin >> a >> b;
		tr.val[a][b] = tr.val[b][a] = 0;
	}
	int res = 0;
	for (int i = 1; i <= n / i; i++) {
		if (n % i == 0) {
			res = (res + dp(tr, i) * phi(n / i)) % P;
			if (i != n / i) {
				res = (res + dp(tr, n / i) * phi(i)) % P;
			}
		}
	}
	res = res * inv(n) % P;
	std::cout << res << '\n';
}

int main() {
	int t; std::cin >> t;
	while (t--) {
		solve();
	}
	return 0;
}
```
