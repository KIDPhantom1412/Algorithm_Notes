## 搜索

### DFS

#### 排列数字

题目：[排列数字](https://www.acwing.com/problem/content/844/)

```c++
#include <cstdio>
#include <cstring>
#include <iostream>
#include <algorithm>

using LL = long long;
using std::cin, std::cout;

const int N = 10;

int n, A[N];
bool st[N];

void dfs(int i)
{
    if (i == n)
    {
        for (int j = 0; j < n; j++) cout << A[j] << ' ';
        cout << '\n';
        return;
    }

    for (int j = 1; j <= n; j++)
        if (!st[j])
        {
            st[j] = true;
            A[i] = j;
            dfs(i + 1);
            st[j] = false;
        }
}

int main()
{
    std::ios::sync_with_stdio(false);
    cin.tie(nullptr), cout.tie(nullptr);

    cin >> n;
    dfs(0);

    return 0;
}
```

#### n-皇后问题

题目：[n-皇后问题](https://www.acwing.com/problem/content/845/)

```c++
#include <cstdio>
#include <cstring>
#include <iostream>
#include <algorithm>
#include <bitset>

using LL = long long;
using std::cin, std::cout;

const int N = 10;

int n;
char board[N][N];
std::bitset<N> col;
std::bitset<N * 2> dg, bdg;

void dfs(int i)
{
    if (i == n)
    {
        for (int j = 0; j < n; j++) cout << board[j] << '\n';
        cout << '\n';
        return;
    }

    for (int j = 0; j < n; j++)
    {
        if (col[j] || dg[i + j] || bdg[j - i + n - 1]) continue;
        col[j] = dg[i + j] = bdg[j - i + n - 1] = true;
        board[i][j] = 'Q';
        dfs(i + 1);
        col[j] = dg[i + j] = bdg[j - i + n - 1] = false;
        board[i][j] = '.';
    }
}

int main()
{
    std::ios::sync_with_stdio(false);
    cin.tie(nullptr), cout.tie(nullptr);

    cin >> n;
    for (int i = 0; i < n; i++)
        for (int j = 0; j < n; j++)
            board[i][j] = '.';
    dfs(0);

    return 0;
}
```



### BFS

#### 走迷宫

题目：[走迷宫](https://www.acwing.com/problem/content/846/)

```c++
#include <cstdio>
#include <cstring>
#include <iostream>
#include <algorithm>

using LL = long long;
using std::cin, std::cout;

const int N = 110;

int n, m;
int G[N][N];

struct Point
{
    int x, y;
};

Point Q[N * N];
bool st[N][N];
int dist[N][N];
int dx[4] = {-1, 0, 1, 0};
int dy[4] = {0, 1, 0, -1};

int bfs()
{
    int hh = 0, tt = 0;
    Q[tt++] = {0, 0};

    while (hh < tt)
    {
        auto [x, y] = Q[hh++];

        for (int i = 0; i < 4; i++)
        {
            int nx = x + dx[i], ny = y + dy[i];
            if (nx < 0 || nx >= n || ny < 0 || ny >= m) continue;
            if (G[nx][ny] == 1 || st[nx][ny]) continue;
            st[nx][ny] = true;
            dist[nx][ny] = dist[x][y] + 1;
            Q[tt++] = {nx, ny};
        }
    }

    return dist[n - 1][m - 1];
}

int main()
{
    std::ios::sync_with_stdio(false);
    cin.tie(nullptr), cout.tie(nullptr);

    cin >> n >> m;
    for (int i = 0; i < n; i++)
        for (int j = 0; j < m; j++)
            cin >> G[i][j];

    cout << bfs() << '\n';
    return 0;
}
```

#### 八数码

题目：[八数码](https://www.acwing.com/problem/content/description/847/)

```c++
#include <cstdio>
#include <cstring>
#include <iostream>
#include <algorithm>
#include <queue>
#include <unordered_map>

using LL = long long;
using std::cin, std::cout;
using std::string;

string start(9, 0), end = "12345678x";
std::unordered_map<string, int> dist;
std::queue<string> Q;

int dx[4] = {-1, 0, 1, 0};
int dy[4] = {0, 1, 0, -1};

int bfs()
{
    if (start == end) return 0;

    Q.push(start);
    dist[start] = 0;

    while (!Q.empty())
    {
        string s = Q.front(); Q.pop();
        int pre = s.find('x'), d = dist[s];
        int px = pre / 3, py = pre % 3;
        for (int i = 0; i < 4; i++)
        {
            int x = px + dx[i], y = py + dy[i];
            if (x >= 0 && x < 3 && y >= 0 && y < 3)
            {
                int cur = x * 3 + y;
                std::swap(s[pre], s[cur]);
                if (!dist.count(s))
                {
                    if (s == end) return d + 1;
                    dist[s] = d + 1;
                    Q.push(s);
                }
                std::swap(s[pre], s[cur]);
            }
        }
    }

    return -1;
}

int main()
{
    std::ios::sync_with_stdio(false);
    cin.tie(nullptr), cout.tie(nullptr);

    for (int i = 0; i < 9; i++) cin >> start[i];
    cout << bfs() << '\n';

    return 0;
}
```



## 图的遍历

### 树与图的深度优先遍历

时间复杂度为$O(n+m)$，$n$表示点数，$m$表示边数

#### 树的重心

题目：[树的重心](https://www.acwing.com/problem/content/848/)

```c++
#include <iostream>
#include <algorithm>

const int N = int(1e5) + 5;

int n, H[N];
struct EdgeNode
{
    int v, n;
} E[N * 2];
int idx;

void add(int a, int b)
{
    E[++idx] = {b, H[a]}, H[a] = idx;
}

int res = N;

int dfs(int u, int fa)
{
    int mx = 0, sum = 1;
    for (int i = H[u]; i; i = E[i].n)
        if (int v = E[i].v; v != fa)
        {
            int t = dfs(v, u);
            sum += t;
            mx =  std::max(mx, t);
        }
    mx = std::max(mx, n - sum);
    res = std::min(res, mx);
    return sum;
}

int main()
{
    std::cin >> n;
    for (int i = 0; i < n - 1; i++)
    {
        int a, b; std::cin >> a >> b;
        add(a, b), add(b, a);
    }
    dfs(1, -1);
    std::cout << res << '\n';
    
    return 0;
}
```

### 树与图的广度优先遍历

时间复杂度为$O(n+m)$，$n$表示点数，$m$表示边数

#### 图中点的层次

题目：[图中点的层次](https://www.acwing.com/problem/content/849/)

```c++
#include <iostream>
#include <algorithm>

const int N = int(1e5) + 5;

int n, m;
int H[N];
struct EdgeNode
{
	int v, n;
} E[N];
int idx;

void add(int a, int b)
{
	E[++idx] = { b, H[a] }, H[a] = idx;
}

int bfs(int start, int end)
{
	if (start == end) return 0;

	static int dist[N]; std::fill_n(dist, N, N);
	static int Q[N];
	int hh = 0, tt = 0;
	dist[start] = 0;
	Q[tt++] = start;
	while (hh != tt)
	{
		int u = Q[hh++];
		for (int i = H[u]; i; i = E[i].n)
		{
			int v = E[i].v;
			if (dist[v] == N)
			{
				dist[v] = dist[u] + 1;
				if (v == end) return dist[v];
				Q[tt++] = v;
			}
		}
	}
	return -1;
}

int main()
{
	std::cin >> n >> m;
	for (int i = 0; i < m; i++)
	{
		int a, b; std::cin >> a >> b;
		add(a, b);
	}

	std::cout << bfs(1, n) << '\n';
	return 0;
}
```

## 拓扑排序

时间复杂度为$O(n+m)$，$n$表示点数，$m$表示边数

### 有向图的拓扑序列

题目：[有向图的拓扑序列](https://www.acwing.com/problem/content/850/)

```c++
#include <iostream>

const int N = int(1e5) + 5;

int n, m;
int H[N];
struct EdgeNode
{
	int v, n;
} E[N];
int idx;

void add(int a, int b)
{
	E[++idx] = { b, H[a] }, H[a] = idx;
}

int inDegree[N];
int Q[N];

bool topSort()
{
	int hh = 0, tt = 0;
	for (int i = 1; i <= n; i++)
		if (inDegree[i] == 0)
		{
			Q[tt++] = i;
		}

	while (hh != tt)
	{
		int u = Q[hh++];
		for (int i = H[u]; i; i = E[i].n)
		{
			int v = E[i].v;
			if (--inDegree[v] == 0)
			{
				Q[tt++] = v;
			}
		}
	}
	return tt == n;
}

int main()
{
	std::cin >> n >> m;
	for (int i = 0; i < m; i++)
	{
		int a, b; std::cin >> a >> b;
		add(a, b);
		inDegree[b]++;
	}
	if (topSort())
	{
		for (int i = 0; i < n; i++) std::cout << Q[i] << ' ';
		std::cout << '\n';
	}
	else
	{
		std::cout << -1 << '\n';
	}
	return 0;
}
```

## 最短路

### Dijkstra

#### 朴素Dijkstra

时间复杂度为$O(n^2+m)$，$n$表示点数，$m$表示边数

##### Dijkstra求最短路 I

题目：[Dijkstra求最短路 I](https://www.acwing.com/problem/content/851/)

题解：https://www.acwing.com/solution/content/38318/

```c++
#include <iostream>
#include <algorithm>

using LL = long long;
using std::cin, std::cout;

const int N = 510, M = int(1e5) + 5, INF = int(1e9) + 10;

int n, m;
int H[N];
struct EdgeNode
{
    int v, w, n;
} E[M];
int idx;

void add(int a, int b, int w)
{
    E[++idx] = { b, w, H[a] }, H[a] = idx;
}

bool st[N];
int dist[N];

void dijkstra()
{
    std::fill_n(dist, N, INF);
    dist[1] = 0;
    for (int i = 0; i < n; i++)
    {
        int u = -1;
        for (int j = 1; j <= n; j++)
            if (!st[j] && (u == -1 || dist[j] < dist[u]))
                u = j;

        st[u] = true;
        if (u == n) return;  // 小小的优化
        
        for (int j = H[u]; j; j = E[j].n)
        {
            int v = E[j].v, w = E[j].w;
            if (!st[v])
            {
                dist[v] = std::min(dist[v], dist[u] + w);
            }
            //else std::cout << -1 << '\n';
        }
    }
}

int main()
{
    std::cin >> n >> m;
    for (int i = 0; i < m; i++)
    {
        int a, b, w; std::cin >> a >> b >> w;
        add(a, b, w);
    }
    
    dijkstra();
    std::cout << (dist[n] == INF ? -1 : dist[n]) << '\n';

    return 0;
}
```

#### 堆优化版Dijkstra

用邻接表存储的图的时间复杂度为$O(m \log_2 n)$，$n$表示点数，$m$表示边数。如果改用邻接矩阵存储图，那么时间复杂度会变为$O(mn)$。

##### Dijkstra求最短路 II

题目：[Dijkstra求最短路 II](https://www.acwing.com/problem/content/852/)

```c++
#include <iostream>
#include <algorithm>
#include <queue>

const int N = int(15e4) + 5, INF = int(1e9) + 5;

int n, m;
int H[N];
struct EdgeNode
{
	int v, w, n;
} E[N];
int idx;

void add(int a, int b, int w)
{
	E[++idx] = { b, w, H[a] }, H[a] = idx;
}

bool st[N];
int dist[N];

void dijkstra()
{
	std::fill_n(dist, N, INF);
	dist[1] = 0;
	struct Node
	{
		int d, u;
		bool operator< (const Node& ct) const
		{
			return d >= ct.d;
		}
	};
	std::priority_queue<Node> Q;
	Q.push({ 0, 1 });
	while (!Q.empty())
	{
		auto [d, u] = Q.top(); Q.pop();
		if (st[u]) continue;

		st[u] = true;
		if (u == n) return;

		for (int i = H[u]; i; i = E[i].n)
		{
			int v = E[i].v, w = E[i].w;
			if (d + w < dist[v])
			{
				dist[v] = d + w;
				Q.push({ dist[v], v });
			}
		}
	}
}

int main()
{
	std::cin >> n >> m;
	for (int i = 0; i < m; i++)
	{
		int a, b, w; std::cin >> a >> b >> w;
		add(a, b, w);
	}

	dijkstra();
	std::cout << (dist[n] == INF ? -1 : dist[n]) << '\n';

	return 0;
}
```

### bellman-ford

时间复杂度$O(nm)$，$n$表示点数，$m$表示边数

#### 有边数限制的最短路

题目：[有边数限制的最短路](https://www.acwing.com/problem/content/855/)

题解：https://www.acwing.com/solution/content/14088/

```c++
#include <iostream>
#include <algorithm>

const int N = 510, M = int(1e4) + 5, INF = int(1e8) + 5;

int n, m;
int H[N];
struct Edge
{
	int a, b, w;
} E[M];

int dist[N], back[N];

void bellmanFord(int start, int K)
{
	std::fill_n(dist + 1, n, INF);
	dist[start] = 0;

	for (int i = 0; i < K; i++)
	{
		std::copy(dist + 1, dist + 1 + n, back + 1);
		for (int j = 0; j < m; j++)
		{
			auto [a, b, w] = E[j];
			dist[b] = std::min(dist[b], back[a] + w);
		}
	}
}

int main()
{
	int K; std::cin >> n >> m >> K;
	for (int i = 0; i < m; i++)
	{
		int a, b, w; std::cin >> a >> b >> w;
		E[i] = { a, b, w };
	}

	bellmanFord(1, K);

	if (dist[n] > INF / 2) std::cout << "impossible" << '\n';
	else std::cout << dist[n] << '\n';

	return 0;
}
```

1. 在迭代过程中，可能会更新`dist[n] = min(dist[n], back[i] + w)`，如果`back[i] = INF`和`w < 0`，则可能出现`dist[n] < INF`的情况。但在理论上`d[n]`依然是正无穷。
1. bf算法在各方面都不如spfa，但题中的应用却只有bf算法才能胜任。

### spfa

时间复杂度平均情况下$O(m)$，最坏情况下$O(nm)$，$n$表示点数，$m$表示边数

#### spfa求最短路

**模板题：**

题目：[spfa求最短路](https://www.acwing.com/problem/content/853/)

题解：https://www.acwing.com/solution/content/105508/

```c++
#include <climits>
#include <iostream>

const int N = int(1e5) + 5;

int n, m;
int H[N];
struct EdgeNode
{
	int v, w, n;
} E[N];
int idx;

void add(int a, int b, int w)
{
	E[++idx] = { b, w, H[a] }, H[a] = idx;
}

bool st[N];
int dist[N];
int Q[N];

void spfa(int start)
{
	std::fill_n(dist + 1, n, INT_MAX);
	dist[start] = 0;
	int hh = 0, tt = 0;
	Q[tt++] = 1;

	while (hh != tt)
	{
		int u = Q[hh++];
		st[u] = false;

		for (int i = H[u]; i; i = E[i].n)
		{
			int v = E[i].v, w = E[i].w;
			if (dist[u] + w < dist[v])
			{
				dist[v] = dist[u] + w;
				if (!st[v])
				{
					Q[tt++] = v;
					st[v] = true;
				}
			}
		}
	}
}

int main()
{
	std::cin >> n >> m;
	for (int i = 0; i < m; i++)
	{
		int a, b, w; std::cin >> a >> b >> w;
		add(a, b, w);
	}

	spfa(1);

	if (dist[n] > INT_MAX / 2) std::cout << "impossible" << '\n';
	else std::cout << dist[n] << '\n';

	return 0;
}
```

#### spfa判断负环

**模板题：**

题目：[spfa判断负环](https://www.acwing.com/problem/content/854/)

```c++
#include <iostream>
#include <algorithm>

const int N = 2010, M = int(1e4) + 5;

int n, m;
int H[N];
struct EdgeNode
{
	int v, w, n;
} E[M];
int idx;

void add(int a, int b, int w)
{
	E[++idx] = { b, w, H[a] }, H[a] = idx;
}

bool st[N];
int Q[N];
int dist[N];
int cnt[N];

bool spfa()
{
	int hh = 0, tt = 0;
	for (int i = 1; i <= n; i++)
	{
		Q[tt++] = i;
		st[i] = true;
	}

	while (hh != tt)
	{
		int u = Q[hh++]; hh %= N;
		st[u] = false;

		for (int i = H[u]; i; i = E[i].n)
		{
			int v = E[i].v, w = E[i].w;
			if (dist[u] + w < dist[v])
			{
				dist[v] = dist[u] + w;
				cnt[v] = cnt[u] + 1;
				if (cnt[v] >= n) return true;
				if (!st[v])
				{
					Q[tt++] = v; tt %= N;
					st[v] = true;
				}
			}
		}
	}
	return false;
}

int main()
{
	std::cin >> n >> m;
	for (int i = 0; i < m; i++)
	{
		int a, b, w; std::cin >> a >> b >> w;
		add(a, b, w);
	}

	std::cout << (spfa() ? "Yes" : "No") << '\n';
	return 0;
}
```

### floyd

时间复杂度$O(n^3)$，$n$表示点数

#### Floyd求最短路

题目：[Floyd求最短路](https://www.acwing.com/problem/content/856/)

题解：https://www.acwing.com/solution/content/49316/

```c++
#include <iostream>

const int N = 210, INF = int(4e6) + 5;

int n, m;
int G[N][N];

void floyd()
{
	for (int k = 1; k <= n; k++)
		for (int i = 1; i <= n; i++)
			for (int j = 1; j <= n; j++)
			{
				G[i][j] = std::min(G[i][j], G[i][k] + G[k][j]);
			}
}

int main()
{
	std::fill_n((int*)G, N * N, INF);
	for (int i = 0; i < N; i++) G[i][i] = 0;
	int t; std::cin >> n >> m >> t;
	for (int i = 0; i < m; i++)
	{
		int a, b, w; std::cin >> a >> b >> w;
		G[a][b] = std::min(G[a][b], w);
	}

	floyd();

	while (t--)
	{
		int a, b; std::cin >> a >> b;
		if (G[a][b] > INF / 2) std::cout << "impossible" << '\n';
		else std::cout << G[a][b] << '\n';
	}
	return 0;
}
```

#### 在线Floyd

题目：https://www.luogu.com.cn/problem/B3647

时间复杂度为$O(mn^2)$

```c++
#include <cstdio>
#include <cstring>
#include <iostream>
#include <algorithm>

using LL = long long;
using std::cin, std::cout;

const int N = 110;

int n, m;
int G[N][N];

void floyd(int a, int b)
{
    for (int i = 1; i <= n; i++)
    {
        G[i][b] = std::min(G[i][b], G[i][a] + G[a][b]);
        G[a][i] = std::min(G[a][i], G[a][b] + G[b][i]);
    }

    for (int i = 1; i <= n; i++)
        for (int j = 1; j <= n; j++)
            G[i][j] = std::min(G[i][j], G[i][a] + G[a][b] + G[b][j]);
}

int main()
{
    std::ios::sync_with_stdio(false);
    cin.tie(nullptr), cout.tie(nullptr);

    memset(G, 0x3f, sizeof G);
    for (int i = 0; i < N; i++) G[i][i] = 0;
    
    cin >> n >> m;
    for (int i = 0; i < m; i++)
    {
        int a, b, w; cin >> a >> b >> w;
        G[a][b] = G[b][a] = std::min(G[a][b], w);
        floyd(a, b), floyd(b, a);
    }

    for (int i = 1; i <= n; i++)
    {
        for (int j = 1; j <= n; j++)
            cout << G[i][j] << ' ';
        cout << '\n';
    }       

    return 0;
}
```

## 最小生成树

### Prim算法求最小生成树

时间复杂度$O(n^2 + m)$，$n$表示点数，$m$表示边数。

题目：[Prim算法求最小生成树](https://www.acwing.com/problem/content/860/)

**邻接矩阵：**

```c++
#include <iostream>
#include <algorithm>

const int N = 510, INF = int(1e8) + 5;

int n, m;
int G[N][N];
bool st[N];
int dist[N];
int res;

bool prim()
{
	std::fill_n(dist + 1, n, INF);
	dist[1] = 0;
	for (int i = 0; i < n; i++)
	{
		int u = -1;
		for (int j = 1; j <= n; j++)
			if (!st[j] && (u == -1 || dist[j] < dist[u]))
				u = j;

		if (dist[u] == INF) return false;

		st[u] = true;
		res += dist[u];

		for (int j = 1; j <= n; j++)
			if (G[u][j] < INF)
			{
				dist[j] = std::min(dist[j], G[u][j]);
			}
	}
	return true;
}

int main()
{
	std::fill_n((int*)G, N * N, INF);
	std::cin >> n >> m;
	for (int i = 0; i < m; i++)
	{
		int a, b, w; std::cin >> a >> b >> w;
		G[a][b] = G[b][a] = std::min(G[a][b], w);
	}
	for (int i = 1; i <= n; i++) G[i][i] = 0;

	if (prim()) std::cout << res << '\n';
	else std::cout << "impossible" << '\n';

	return 0;
}
```

```c++
#include <cstdio>

{
```

### Kruskal算法求最小生成树

时间复杂度$O(m\log_2m)$，$n$表示点数，$m$​表示边数。

Kruskal算法适用于稀疏图，Prim算法适用于稠密图。

题目：[Kruskal算法求最小生成树](https://www.acwing.com/problem/content/861/)

```c++
#include <iostream>
#include <algorithm>

const int N = int(1e5) + 5, M = int(2e5) + 5;

int n, m;
struct Edge
{
	int a, b, w;

	bool operator< (const Edge& ct) const
	{
		return w < ct.w;
	}
} E[M];

int res;
int P[N];

int find(int x)
{
	if (x != P[x]) P[x] = find(P[x]);
	return P[x];
}

bool kruskal()
{
	for (int i = 1; i <= n; i++) P[i] = i;
	std::sort(E, E + m);

	int cnt = 0;
	for (int i = 0; i < m; i++)
	{
		auto [a, b, w] = E[i];
		if (find(a) != find(b))
		{
			P[find(a)] = find(b);
			res += w;
			if (++cnt == n - 1) return true;
		}
	}
	return false;
}

int main()
{
	std::cin >> n >> m;
	for (int i = 0; i < m; i++)
	{
		int a, b, w; std::cin >> a >> b >> w;
		E[i] = { a, b ,w };
	}

	if (kruskal()) std::cout << res << '\n';
	else std::cout << "impossible" << '\n';

	return 0;
}
```

### 染色法判定二分图

时间复杂度$O(n + m)$，$n$表示点数，$m$表示边数。

题目：[染色法判定二分图](https://www.acwing.com/problem/content/862/)

bfs或者dfs都可以做。

```c++
#include <cstdio>
#include <cstring>
#include <iostream>
#include <algorithm>

using LL = long long;
using std::cin, std::cout;

const int N = 1e5 + 5;

int n, m;
int H[N];
struct EdgeNode
{
    int v, n;
} E[N * 2];
int idx;

void add(int a, int b)
{
    E[++idx] = {b, H[a]}, H[a] = idx;
}

int color[N];

bool dfs(int u, int c)
{
    color[u] = c;

    for (int p = H[u]; p; p = E[p].n)
    {
        int v = E[p].v;
        if (color[v] == 0)
        {
            if (!dfs(v, -c)) return false;
        }
        else
        {
            if (color[v] == color[u]) return false;
        }
    }

    return true;
}

int main()
{
    std::ios::sync_with_stdio(false);
    cin.tie(nullptr), cout.tie(nullptr);

    cin >> n >> m;
    for (int i = 0; i < m; i++)
    {
        int a, b; cin >> a >> b;
        add(a, b), add(b, a);
    }

    for (int i = 1; i <= n; i++)
        if (color[i] == 0)
            if (!dfs(i, 1))
            {
                cout << "No" << '\n';
                return 0;
            }

    cout << "Yes" << '\n';
    return 0;
}
```

### 匈牙利算法

时间复杂度$O(nm)$，$n$表示点数，$m$表示边数。

#### 二分图的最大匹配

题目：[二分图的最大匹配](https://www.acwing.com/problem/content/863/)

题解：https://www.acwing.com/solution/content/179030/

```c++
#include <cstdio>
#include <cstring>
#include <iostream>
#include <algorithm>

using LL = long long;
using std::cin, std::cout;

const int N = 510, M = 1e5 + 5;

int n1, n2, m;
int H[N];
struct EdgeNode
{
    int v, n;
} E[M];
int idx;

void add(int a, int b)
{
    E[++idx] = {b, H[a]}, H[a] = idx;
}

int match[N];
bool st[N];

bool find(int u)
{
    for (int p = H[u]; p; p = E[p].n)
    {
        int v = E[p].v;

        if (!st[v])
        {
            st[v] = true;
            if (match[v] == 0 || find(match[v]))
            {
                match[v] = u;
                return true;
            }
        }
    }

    return false;
}

int main()
{
    std::ios::sync_with_stdio(false);
    cin.tie(nullptr), cout.tie(nullptr);

    cin >> n1 >> n2 >> m;
    for (int i = 0; i < m; i++)
    {
        int a, b; cin >> a >> b;
        add(a, b);
    }

    int res = 0;
    for (int i = 1; i <= n1; i++)
    {
        memset(st, 0, sizeof st);
        res += find(i);
    }

    cout << res << '\n';

    return 0;
}
```
